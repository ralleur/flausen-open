cmake_minimum_required(VERSION 3.19)
project(ralleur-desktop-scroll LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Widgets)
find_package(KWin REQUIRED)
add_library(desktopscroll MODULE native/desktopscroll.cpp)
target_link_libraries(desktopscroll PRIVATE Qt6::Core Qt6::Gui Qt6::Qml KWin::kwin)
