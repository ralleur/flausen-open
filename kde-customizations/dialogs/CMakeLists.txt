cmake_minimum_required(VERSION 3.24)
project(RalleurDialogs VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_library(dialogstyle STATIC dialogstyle.cpp dialogstyle.h)
set_target_properties(dialogstyle PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(dialogstyle PUBLIC Qt6::Widgets)
target_compile_options(dialogstyle PRIVATE -Wall -Wextra -Wpedantic)
add_library(ralleurdialogs MODULE plugin.cpp)
target_link_libraries(ralleurdialogs PRIVATE dialogstyle)
set_target_properties(ralleurdialogs PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/styles")
add_executable(dialog_test tests/dialog_test.cpp)
target_link_libraries(dialog_test PRIVATE dialogstyle Qt6::Test)
enable_testing()
add_test(NAME dialogs COMMAND dialog_test -platform offscreen)
add_executable(preview preview.cpp)
target_link_libraries(preview PRIVATE Qt6::Widgets)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME installer_roundtrip COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/install_test.py)
add_executable(menu-preview menu-preview.cpp)
target_link_libraries(menu-preview PRIVATE Qt6::Widgets)

add_executable(slider_test ../sliders/tests/native_test.cpp)
target_link_libraries(slider_test PRIVATE dialogstyle Qt6::Test)
add_test(NAME sliders COMMAND slider_test -platform offscreen)
