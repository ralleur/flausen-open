cmake_minimum_required(VERSION 3.24)
project(RalleurDolphinHeader VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets DBus Network Test Concurrent PdfWidgets MultimediaWidgets)
find_package(KF6WindowSystem REQUIRED)
find_package(KF6KIO REQUIRED)
find_package(KF6XmlGui REQUIRED)
find_package(KF6IconThemes REQUIRED)
find_package(KF6I18n REQUIRED)
# Qt Widgets cannot import the QML singleton. Generate its native material bridge
# from the same source; changing the shared token also triggers reconfiguration.
set(MENU_STYLE "${CMAKE_CURRENT_SOURCE_DIR}/../taskbar/menu-style/MenuStyle.qml")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${MENU_STYLE}")
file(READ "${MENU_STYLE}" MENU_STYLE_SOURCE)
string(REGEX MATCH "readonly property color smokeSurface: Colours.light \\? \"(#[0-9a-fA-F]+)\" : \"(#[0-9a-fA-F]+)\"" SMOKE_TOKEN "${MENU_STYLE_SOURCE}")
if(NOT SMOKE_TOKEN)
    message(FATAL_ERROR "Cannot read shared MenuStyle.smokeSurface")
endif()
set(SMOKE_LIGHT "${CMAKE_MATCH_1}")
set(SMOKE_DARK "${CMAKE_MATCH_2}")
configure_file(smokematerial.h.in smokematerial.h @ONLY)
add_executable(preview_defaults preview_defaults.cpp)
target_link_libraries(preview_defaults PRIVATE Qt6::Core KF6::KIOGui)
add_library(ralleur_header STATIC header.cpp header.h foldercolors.cpp foldercolors.h quickpreview.cpp quickpreview.h contextmenu.cpp contextmenu.h)
set_target_properties(ralleur_header PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(ralleur_header PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(ralleur_header PUBLIC Qt6::Widgets Qt6::DBus Qt6::Network Qt6::Concurrent Qt6::PdfWidgets Qt6::MultimediaWidgets KF6::WindowSystem KF6::KIOFileWidgets KF6::XmlGui KF6::IconThemes KF6::I18n)
target_compile_options(ralleur_header PRIVATE -Wall -Wextra -Wpedantic)
add_library(ralleurdolphin MODULE plugin.cpp)
target_link_libraries(ralleurdolphin PRIVATE ralleur_header)
set_target_properties(ralleurdolphin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/generic")
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME installer_rollback COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/install_test.py)
add_executable(header_test tests/header_test.cpp)
target_link_libraries(header_test PRIVATE ralleur_header Qt6::Test KF6::KIOFileWidgets)
add_test(NAME header_interactions COMMAND header_test -platform offscreen)
add_executable(foldercolors_test tests/foldercolors_test.cpp)
target_link_libraries(foldercolors_test PRIVATE ralleur_header Qt6::Test)
add_test(NAME folder_color_inheritance COMMAND foldercolors_test -platform offscreen)
add_executable(quickpreview_test tests/quickpreview_test.cpp)
target_link_libraries(quickpreview_test PRIVATE ralleur_header Qt6::Test)
add_test(NAME quick_preview_keyboard COMMAND quickpreview_test -platform offscreen)
add_library(preview_probe MODULE EXCLUDE_FROM_ALL tests/live_preview_probe.cpp)
target_link_libraries(preview_probe PRIVATE Qt6::Widgets KF6::XmlGui KF6::IconThemes)
set_target_properties(preview_probe PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/live-probe/generic")
add_executable(plugin_lifecycle tests/plugin_lifecycle.cpp)
target_link_libraries(plugin_lifecycle PRIVATE Qt6::Widgets)
set_target_properties(plugin_lifecycle PROPERTIES OUTPUT_NAME dolphin RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lifecycle")
add_dependencies(plugin_lifecycle ralleurdolphin)
add_test(NAME plugin_process_lifecycle COMMAND ${CMAKE_COMMAND} -E env
  "QT_QPA_GENERIC_PLUGINS=ralleurdolphin" "QT_PLUGIN_PATH=${CMAKE_BINARY_DIR}/plugins"
  "RALLEUR_DOLPHIN_PLUGIN_DIR=${CMAKE_BINARY_DIR}/plugins"
  $<TARGET_FILE:plugin_lifecycle> -platform offscreen)

add_executable(contextmenu_test tests/contextmenu_test.cpp)
target_link_libraries(contextmenu_test PRIVATE ralleur_header Qt6::Test)
add_test(NAME context_menu COMMAND contextmenu_test -platform offscreen)
add_library(context_probe MODULE EXCLUDE_FROM_ALL tests/live_context_probe.cpp)
target_link_libraries(context_probe PRIVATE Qt6::Widgets)
set_target_properties(context_probe PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/live-probe/generic")
