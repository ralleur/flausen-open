cmake_minimum_required(VERSION 3.19)
project(ralleur-tiling-input LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml DBus Widgets)
find_package(KWin REQUIRED)
find_package(KF6Config REQUIRED)
add_library(ralleurtilinginput MODULE native/tilinginput.cpp)
target_link_libraries(ralleurtilinginput PRIVATE Qt6::Core Qt6::Gui Qt6::Qml KWin::kwin)
add_library(ralleurtilingguard MODULE native/shiftguard.cpp)
target_link_libraries(ralleurtilingguard PRIVATE Qt6::Core Qt6::Gui Qt6::Qml KWin::kwin KF6::ConfigCore)

add_library(ralleurtilinglaunch MODULE native/launchbridge.cpp)
target_link_libraries(ralleurtilinglaunch PRIVATE Qt6::Core Qt6::Qml Qt6::DBus)
