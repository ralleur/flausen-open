cmake_minimum_required(VERSION 3.19)
project(ralleur-tiling-probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml DBus Widgets)
find_package(KWin REQUIRED)
add_library(ralleurtilingprobe MODULE probe.cpp)
target_link_libraries(ralleurtilingprobe PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::DBus KWin::kwin)
